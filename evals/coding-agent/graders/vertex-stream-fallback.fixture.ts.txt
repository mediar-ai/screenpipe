  const key = await crypto.subtle.generateKey({name:'RSASSA-PKCS1-v1_5',modulusLength:2048,publicExponent:new Uint8Array([1,0,1]),hash:'SHA-256'},true,['sign','verify']);

  return 'data: '+JSON.stringify({id:'synthetic-stream',model:'synthetic-model',choices:[{index:0,delta,finish_reason}]})+'\n\n';
}
async function stream(text: string, bytewise=false, status=200) {
  resetTokenCache();let requests=0;
  const body:any={model:'glm-4.7',messages:[{role:'user',content:'Synthetic request'}],temperature:0,max_tokens:64};
  const before=JSON.stringify(body);
  globalThis.fetch=(async (input:any,init:RequestInit={}) => {
    const url=String(input);
    if(url==='https://oauth2.googleapis.com/token')return Response.json({access_token:'synthetic-token',expires_in:3600});
    expect(url).toBe('https://aiplatform.googleapis.com/v1/projects/synthetic-project/locations/global/endpoints/openapi/chat/completions');
    expect(init.method).toBe('POST');expect(new Headers(init.headers).get('authorization')).toBe('Bearer synthetic-token');
    expect(JSON.parse(String(init.body))).toMatchObject({messages:body.messages,temperature:0,max_tokens:64,stream:true});requests++;
    const bytes=encoder.encode(text);
    return new Response(new ReadableStream({start(c){if(bytewise){for(const byte of bytes)c.enqueue(Uint8Array.of(byte));}else c.enqueue(bytes);c.close();}}),{status});
  }) as typeof fetch;
  const provider=new VertexMaasProvider(JSON.stringify({project_id:'synthetic-project',client_email:'synthetic@eval.invalid',private_key:pem}),'synthetic-project');
  try{return await new Response(await provider.createStreamingCompletion(body)).text();}
  finally{expect(requests).toBe(1);expect(JSON.stringify(body)).toBe(before);}
}
function parsed(text:string) {
  return text.split('\n').filter(l=>l.startsWith('data: ')&&l.trim()!=='data: [DONE]').flatMap(l=>{try{return [JSON.parse(l.slice(6))];}catch{return [];}});
}
function values(text:string,key:string) {return parsed(text).flatMap(x=>x.choices??[]).map(c=>c.delta?.[key]).filter(x=>typeof x==='string').join('');}

  if(label!=='eof')expect(out.indexOf('"content":')).toBeLessThan(out.indexOf('data: [DONE]'));
});
