import { beforeEach, afterAll, expect, mock, test } from 'bun:test';
let chunks: any[], calls: any[], initialError: Error | null, streamError: any;
mock.module('openai', () => ({ default: class {
  chat = { completions: { create: async (params: any) => {
    calls.push(structuredClone(params));
    if (initialError) throw initialError;
    return { controller: { abort() {} }, async *[Symbol.asyncIterator]() {
      for (const chunk of chunks) yield structuredClone(chunk);
      if (streamError) throw streamError;
    }};
  }}};
}}));
const oldFetch = globalThis.fetch;
globalThis.fetch = (() => { throw new Error('Live network forbidden'); }) as any;
afterAll(() => { globalThis.fetch = oldFetch; });
const { OpenRouterProvider } = await import('./src/providers/openrouter');
const request = () => ({ model: 'synthetic/model', messages: [{ role: 'user', content: 'Synthetic question' }], temperature: 0.3, tools: [{ type: 'function', function: { name: 'lookup', parameters: { type: 'object' } } }] });
const chunk = (delta: any, extra: any = {}) => ({ id: 'synthetic-id', object: 'chat.completion.chunk', created: 1700000000, model: 'synthetic/model', choices: [{ index: 0, delta, finish_reason: null }], ...extra });
beforeEach(() => { chunks = []; calls = []; initialError = null; streamError = null; });
async function collect(input: any = request()) {
  const stream = await new OpenRouterProvider('synthetic-key').createStreamingCompletion(input);
  const raw = await new Response(stream).text();
  const events = raw.split(/\r?\n\r?\n/).map(block => block.split(/\r?\n/).filter(line => line.startsWith('data:')).map(line => line.slice(5).trimStart()).join('\n')).filter(Boolean);
  expect(events.filter(x => x === '[DONE]')).toHaveLength(1);
  expect(events.at(-1)).toBe('[DONE]');
  return events.filter(x => x !== '[DONE]').map(x => JSON.parse(x));
}
test('text chunks retain response identity and choice index', async () => {
  chunks = [chunk({ content: 'hello' })];
  expect((await collect())[0]).toMatchObject({ id: 'synthetic-id', object: 'chat.completion.chunk', created: 1700000000, model: 'synthetic/model', choices: [{ index: 0, delta: { content: 'hello' } }] });
});
test('assistant role-only chunk survives', async () => {
  chunks = [chunk({ role: 'assistant' })];
  expect((await collect())[0]?.choices[0].delta.role).toBe('assistant');
});
test('tool-only fragments retain identity name and ordered arguments', async () => {
  chunks = [chunk({ tool_calls: [{ index: 0, id: 'tool-a', type: 'function', function: { name: 'lookup', arguments: '{"query":' } }] }), chunk({ tool_calls: [{ index: 0, function: { arguments: '"safe Ω"}' } }] })];
  const events = await collect();
  expect(events[0]?.choices[0].delta.tool_calls[0]).toMatchObject({ index: 0, id: 'tool-a', type: 'function', function: { name: 'lookup' } });
  expect(events.map(e => e.choices[0].delta.tool_calls[0].function.arguments).join('')).toBe('{"query":"safe Ω"}');
});
test('finish-only delta retains finish reason', async () => {
  chunks = [chunk({}, { choices: [{ index: 0, delta: {}, finish_reason: 'tool_calls' }] })];
  expect((await collect())[0]?.choices[0]).toMatchObject({ index: 0, delta: {}, finish_reason: 'tool_calls' });
});
test('usage-only chunk retains zero and nonzero counts', async () => {
  chunks = [chunk({}, { choices: [], usage: { prompt_tokens: 0, completion_tokens: 17, total_tokens: 17 } })];
  expect((await collect())[0]?.usage).toEqual({ prompt_tokens: 0, completion_tokens: 17, total_tokens: 17 });
});
test('explicit empty content is not discarded', async () => {
  chunks = [chunk({ content: '' })];
  expect((await collect())[0]?.choices[0].delta.content).toBe('');
});
test('multiple choices retain their own indices and payloads', async () => {
  chunks = [chunk({}, { choices: [{ index: 2, delta: { content: 'second' }, finish_reason: null }, { index: 7, delta: { role: 'assistant' }, finish_reason: 'stop' }] })];
  expect((await collect())[0]?.choices).toEqual(chunks[0].choices);
});
test('ordinary Unicode text keeps order and terminates once', async () => {
  chunks = [chunk({ content: 'Hello 🌿' }), chunk({ content: ' Ω world' })];
  expect((await collect()).map(e => e.choices[0].delta.content).join('')).toBe('Hello 🌿 Ω world');
});
test('empty upstream stream terminates without fabricated chunks', async () => { expect(await collect()).toEqual([]); });
test('one upstream request preserves generation inputs and caller state', async () => {
  const body = request(), original = structuredClone(body); await collect(body);
  expect(calls).toHaveLength(1); expect(calls[0]).toMatchObject({ ...original, stream: true }); expect(body).toEqual(original);
});
test('partial text survives iterator error with error detail and termination', async () => {
  chunks = [chunk({ content: 'retained' })]; streamError = Object.assign(new Error('synthetic refusal'), { status: 429, error: { type: 'rate_limit' } });
  const events = await collect(); expect(events[0]?.choices[0].delta.content).toBe('retained');
  expect(events.at(-1)?.error).toMatchObject({ message: 'synthetic refusal', type: 'rate_limit', code: '429' });
});
test('initial provider rejection remains a rejected call', async () => {
  initialError = new Error('initial refusal'); await expect(collect()).rejects.toThrow('initial refusal'); expect(calls).toHaveLength(1);
});
