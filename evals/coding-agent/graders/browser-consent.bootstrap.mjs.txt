import {spawnSync} from 'node:child_process';
const root=process.cwd(),ports=join(root,'.eval-browser-consent-ports'),config=join(root,'.eval-browser-consent.config.mjs');
const mocked=['@/lib/chat-storage','@/lib/browser-state-cache','@/lib/hooks/use-settings','@/lib/utils/tauri','@/lib/api','@/components/file-preview-sidebar','@/components/right-panel-tab-strip','@/lib/i18n/provider','@/components/chat/chat-panel-home','@/components/ui/button','@tauri-apps/api/event','@tauri-apps/api/dpi','@tauri-apps/api/menu','@tauri-apps/api/window','@tauri-apps/plugin-opener','@tauri-apps/plugin-os','gt-react'];
const alias={};
try{
 mkdirSync(ports,{recursive:true});
 for(const [index,name]of mocked.entries()){const file=join(ports,String(index)+'.ts');writeFileSync(file,'export {};');alias[name]=file}
 alias['@']=root;
 const options={resolve:{alias},esbuild:{jsx:'automatic'},css:{postcss:{plugins:[]}},test:{environment:'jsdom',include:['components/.eval-browser-consent.test.tsx'],pool:'forks',poolOptions:{forks:{singleFork:true}}}};

}finally{rmSync(ports,{recursive:true,force:true});rmSync(config,{force:true})}
