function emit(event: string, payload: unknown): void {
  mocks.listeners.get(event)?.({ payload });
}

vi.mock("gt-react",()=>({useGT:()=> (text: string)=>text}));
vi.mock("@/lib/i18n/provider",()=>({useUiLocale:()=>"en"}));
vi.mock("@/components/chat/chat-panel-home",()=>({ChatPanelHome:()=>null}));

describe("BrowserSidebar consent outcomes", () => {
  beforeEach(() => {
    mocks.platform = "macos";
    vi.stubGlobal("fetch",()=>{throw Error("Live network forbidden")});
    mocks.setBrowserCookieAccessState.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserResolveSessionAccess.mockReset().mockResolvedValue(undefined);
    mocks.listeners.clear();
    mocks.listen
      .mockReset()
      .mockImplementation(async (event: string, listener: TauriListener) => {
        mocks.listeners.set(event, listener);
        return () => mocks.listeners.delete(event);
      });
    mocks.loadConversationFile.mockReset().mockResolvedValue(null);
    mocks.updateConversationFlags.mockReset().mockResolvedValue(undefined);
    mocks.updateSettings.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserHide.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserTabHide.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserNavigate.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserTabNavigate.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserTabClose.mockReset().mockResolvedValue(undefined);
    mocks.ownedBrowserSetBounds.mockReset().mockResolvedValue(undefined);
    vi.stubGlobal("ResizeObserver", ResizeObserverMock);
    vi.spyOn(HTMLElement.prototype, "getBoundingClientRect").mockReturnValue({
      x: 0,
      y: 0,
      top: 0,
      left: 0,
      right: 480,
      bottom: 640,
      width: 480,
      height: 640,
      toJSON: () => ({}),
    });
    vi.spyOn(HTMLElement.prototype, "clientWidth", "get").mockReturnValue(1200);
  });

  afterEach(() => {
    cleanup();
    vi.restoreAllMocks();
    vi.unstubAllGlobals();
  });


  for(const field of ['navigation_id','navigationId'])for(const userTab of [false,true])for(const allow of [false,true])it(`consent ${field} userTab=${userTab} allow=${allow}`,async()=>{
    render(<BrowserSidebar conversationId="synthetic-chat"/>);
    let tabId:string|undefined;
    if(userTab){act(()=>{window.dispatchEvent(new CustomEvent('screenpipe:browser-sidebar-new-tab'))});tabId=screen.getByTestId(/^tab-loading-tab-/).getAttribute('data-testid')!.replace('tab-loading-','')}
    act(()=>{emit('owned-browser:navigate',{url:'https://example.test/',owner:'synthetic-chat',navigationId:'current-nav',reveal:true,...(tabId?{tabId}:{})});emit('owned-browser:session-access-request',{request_id:'synthetic-request',url:'https://example.test/',host:'example.test',owner:'synthetic-chat',[field]:'current-nav',already_granted:false})});
    expect(screen.getByRole('button',{name:'Continue logged out'})).toBeVisible();
    expect(screen.queryByRole('progressbar')).toBeNull();
    expect(mocks.ownedBrowserResolveSessionAccess).not.toHaveBeenCalled();expect(mocks.setBrowserCookieAccessState).not.toHaveBeenCalled();expect(mocks.updateSettings).not.toHaveBeenCalled();
    if(tabId)expect(mocks.ownedBrowserTabHide).toHaveBeenCalledWith(tabId);
    fireEvent.click(screen.getByRole('button',{name:allow?'Use browser session':'Continue logged out'}));
    await waitFor(()=>expect(mocks.ownedBrowserResolveSessionAccess).toHaveBeenCalledWith('synthetic-request',allow));
    expect(mocks.ownedBrowserResolveSessionAccess).toHaveBeenCalledTimes(1);expect(mocks.setBrowserCookieAccessState).toHaveBeenCalledWith(allow,!allow);expect(mocks.updateSettings).toHaveBeenCalledWith({browserCookieAccessGranted:allow});
    await waitFor(()=>expect(screen.queryByRole('button',{name:'Continue logged out'})).toBeNull());
  });
  for(const [label,owner,nav] of [['stale','synthetic-chat','old'],['missing navigation','synthetic-chat',undefined],['foreign','other-chat','current-nav'],['ownerless',undefined,'current-nav']])it(`rejects ${label}`,()=>{
    render(<BrowserSidebar conversationId="synthetic-chat"/>);
    act(()=>{emit('owned-browser:navigate',{url:'https://example.test/',owner:'synthetic-chat',navigationId:'current-nav',reveal:true});emit('owned-browser:session-access-request',{request_id:'ignored',url:'https://example.test/',host:'example.test',owner,navigation_id:nav,already_granted:false})});
    expect(screen.queryByRole('button',{name:'Continue logged out'})).toBeNull();expect(mocks.ownedBrowserResolveSessionAccess).not.toHaveBeenCalled();expect(mocks.setBrowserCookieAccessState).not.toHaveBeenCalled();expect(mocks.updateSettings).not.toHaveBeenCalled();
  });
});
