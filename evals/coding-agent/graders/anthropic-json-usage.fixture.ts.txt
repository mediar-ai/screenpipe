import { beforeEach, afterAll, expect, mock, test } from 'bun:test';
let response: any;
let rejection: Error | null;
let calls: any[];
mock.module('@anthropic-ai/sdk', () => ({ default: class {
  messages = { create: async (params: any) => {
    calls.push(structuredClone(params));
    if (rejection) throw rejection;
    return structuredClone(response);
  }};
}}));
const originalFetch = globalThis.fetch;
globalThis.fetch = (() => { throw new Error('External network is forbidden in this fixture'); }) as any;
afterAll(() => { globalThis.fetch = originalFetch; });
const { AnthropicProvider } = await import('./src/providers/anthropic');
const body = () => ({ model: 'claude-synthetic@20260101', messages: [{ role: 'user', content: 'Synthetic question' }], max_tokens: 97, temperature: 0.2 });
beforeEach(() => {
  calls = []; rejection = null;
  response = { id: 'synthetic-response', role: 'assistant', type: 'message', content: [{ type: 'text', text: 'Synthetic answer 🌿' }], usage: { input_tokens: 29, output_tokens: 7 }, stop_reason: 'end_turn' };
});
async function complete(provider = new AnthropicProvider('synthetic-key'), request: any = body()) {
  const res = await provider.createCompletion(request);
  expect(res.status).toBe(200);
  expect(res.headers.get('content-type')).toContain('application/json');
  return res.json() as Promise<any>;
}
for (const [input, output] of [[0, 0], [0, 23], [57, 0], [1234, 56], [1000003, 4567], [19, 31]]) {
  test(`reported uncached usage survives JSON completion: ${input}/${output}`, async () => {
    response.usage = { input_tokens: input, output_tokens: output };
    const result = await complete();
    expect(result.usage).toMatchObject({ prompt_tokens: input, completion_tokens: output, total_tokens: input + output });
    expect(calls).toHaveLength(1);
  });
}
test('successive responses retain their own usage', async () => {
  const provider = new AnthropicProvider('synthetic-key');
  for (const [input, output] of [[41, 11], [3, 101], [0, 0]]) {
    response.usage = { input_tokens: input, output_tokens: output };
    expect((await complete(provider)).usage).toMatchObject({ prompt_tokens: input, completion_tokens: output, total_tokens: input + output });
  }
  expect(calls).toHaveLength(3);
});
test('visible Unicode answer and assistant role remain intact', async () => {
  expect((await complete()).choices[0].message).toMatchObject({ content: 'Synthetic answer 🌿', role: 'assistant' });
});
test('tool names and structured arguments survive beside text', async () => {
  response.content.push({ type: 'tool_use', id: 'synthetic-tool', name: 'lookup_fixture', input: { query: 'safe Ω', nested: { limit: 2 } } });
  const message = (await complete()).choices[0].message;
  expect(message.content).toBe('Synthetic answer 🌿');
  expect(message.tool_calls).toHaveLength(1);
  expect(message.tool_calls[0].function.name).toBe('lookup_fixture');
  expect(JSON.parse(message.tool_calls[0].function.arguments)).toEqual({ query: 'safe Ω', nested: { limit: 2 } });
});
test('tool-only responses preserve empty text and each tool', async () => {
  response.content = ['first', 'second'].map((name, i) => ({ type: 'tool_use', id: `synthetic-${i}`, name, input: { n: i } }));
  const message = (await complete()).choices[0].message;
  expect(message.content).toBe('');
  expect(message.tool_calls.map((x: any) => [x.function.name, JSON.parse(x.function.arguments)])).toEqual([['first', { n: 0 }], ['second', { n: 1 }]]);
});
test('one upstream request preserves model and generation inputs', async () => {
  const request = body(); const original = structuredClone(request);
  await complete(undefined, request);
  expect(calls).toHaveLength(1);
  expect(calls[0]).toMatchObject({ model: 'claude-synthetic-20260101', max_tokens: 97, temperature: 0.2, messages: [{ role: 'user', content: [{ type: 'text', text: 'Synthetic question' }] }] });
  expect(request).toEqual(original);
});
test('upstream rejection remains a failure without a fabricated completion', async () => {
  rejection = new Error('synthetic provider refusal');
  await expect(complete()).rejects.toThrow();
  expect(calls).toHaveLength(1);
});
